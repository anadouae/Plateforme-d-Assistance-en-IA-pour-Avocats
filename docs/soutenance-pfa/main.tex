\section{Cadrage \& Problématique}

\begin{frame}{Contexte métier des cabinets d'avocats}
\framesubtitle{De l'oralité d'audience à la production écrite sous contrainte}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.54\textwidth}
  \textbf{Le quotidien d'un cabinet marocain :}
  \begin{itemize}
    \item Production documentaire volumineuse : requêtes, mémoires, conclusions et notes de plaidoirie.
    \item Rythme imposé par les audiences judiciaires.
    \item Environnement bilingue (\textbf{arabe standard} devant les juridictions, français en affaires).
  \end{itemize}

  \smallskip
  \begin{alertblock}{Le circuit artisanal traditionnel}
    Notes d'audience $\rightarrow$ dictée informelle (WhatsApp) $\rightarrow$ ressaisie par collaborateur $\rightarrow$ mise en page Word tardive.
    \begin{itemize}
      \item \textbf{Perte de temps} et risque d'erreurs matérielles.
      \item \textbf{Aucune traçabilité} centralisée du dossier.
    \end{itemize}
  \end{alertblock}
\end{column}

\begin{frame}{Objectifs et positionnement de LexHub}
\framesubtitle{Un copilote déontologique, pas un « robot-avocat »}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
  \textbf{Objectifs majeurs du projet :}
  \begin{itemize}
    \item \textbf{Pipeline en 5 étapes :} de la voix et des pièces vers le document finalisé.
    \item \textbf{Reconnaissance vocale native :} via Web Speech API, zéro coût récurrent de transcription serveur.
    \item \textbf{Export Word (.docx) unique :} prêt pour l'audience avec armoiries officielles.
    \item \textbf{Isolation multi-tenant stricte :} par cabinet.
    \item \textbf{Modèle économique pérenne :} forfaits, essai hybride, packs prépayés.
  \end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
  \begin{exampleblock}{Principe d'ancrage factuel strict}
    \begin{itemize}
      \item L'IA structure, formule et polit en arabe juridique (\textit{Fusha}).
      \item \alert{Interdiction formelle} d'inventer des faits, des articles de loi ou des montants non dictés.
      \item L'avocat demeure seul responsable du contenu plaidé ou déposé.
    \end{itemize}
  \end{exampleblock}

\begin{frame}{Étape 1 \& 2 : Dossier, Pièces \& Dictée Vocale}
\framesubtitle{Capture de la parole via la Web Speech API native}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
  \textbf{Étape 1 — Dossier et pièces jointes :}
  \begin{itemize}
    \item Rattachement au dossier judiciaire (tribunal, numéro d'ordre, parties).
    \item Intégration des pièces justificatives (contrats, PV d'huissier, factures).
    \item Qualification du domaine (civil, commercial, famille, pénal).
  \end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
  \textbf{Étape 2 — Dictée vocale native :}
  \begin{itemize}
    \item Standard W3C \textbf{Web Speech API} intégré au navigateur.
    \item Affichage en direct des mots prononcés.
  \end{itemize}

  \smallskip
  \begin{exampleblock}{Avantages technologiques}
    \begin{itemize}
      \item \textbf{Latence nulle :} flux audio traité localement.
      \item \textbf{Confidentialité :} pas d'envoi d'enregistrements audio bruts vers des serveurs tiers.
      \item \textbf{Coût zéro :} aucune facture API Whisper serveur.
    \end{itemize}
  \end{exampleblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Étape 3 : Transcription Fusha \& Détection d'erreurs}
\framesubtitle{Nettoyage typographique et contrôle déterministe des contradictions}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
  \textbf{Correction en arabe standard (Fusha) :}
  \begin{itemize}
    \item Normalisation de la syntaxe et du vocabulaire juridique marocain.
    \item \textbf{Surlignage visuel orange} des termes rectifiés : l'avocat garde le contrôle visuel immédiat sur chaque retouche.
  \end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
  \begin{alertblock}{Audit déterministe des contradictions}
    Contrôle algorithmique sans hallucination sur 4 éléments matériels critiques :
    \begin{enumerate}
      \item \textbf{Dates :} cohérence chronologique des faits.
      \item \textbf{Numéros de pièces :} concordance avec les annexes.
      \item \textbf{Montants :} sommes réclamées en dirhams (MAD).
      \item \textbf{Identités :} demandeur vs défendeur.
    \end{enumerate}
  \end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Étape 4 : Rédaction Assistée \& Révision Interactive}
\framesubtitle{Mise en forme selon les standards des tribunaux marocains}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
  \textbf{Structure procédurale normalisée :}
  \begin{itemize}
    \item \textbf{En-tête de cour :} juridiction, chambre, parties.
    \item \textbf{Faits constants (Waqā'i') :} chronologie fidèle.
    \item \textbf{Moyens de droit (Asbāb) :} réfutation des thèses adverses, visas légaux.
    \item \textbf{Dispositif légal (Talabāt) :} prétentions chiffrées précises.
  \end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
  \begin{block}{Assistant de révision par dialogue (Chat)}
    \begin{itemize}
      \item Dialogue interactif pour affiner un argument spécifique sans régénérer tout le document.
      \item Prise en compte de requêtes ciblées (\textit{« Développe l'argument sur le défaut de livraison »}).
    \end{itemize}
  \end{block}

  \smallskip
  \begin{center}
    \badgegreen{Ancrage factuel} \quad \badgeblue{Zéro invention}
  \end{center}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Étape 5 : Export Officiel Microsoft Word (.docx) Unique}
\framesubtitle{Un livrable 100\,\% éditable, prêt pour l'audience}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
  \textbf{Pourquoi Word (.docx) exclusif ?}
  \begin{itemize}
    \item Le PDF figé est inadapté : l'avocat doit pouvoir ajuster une ligne à la volée avant de plaider.
    \item Compatibilité universelle avec le secrétariat et les greffes des tribunaux.
  \end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
  \begin{exampleblock}{Finition professionnelle du document}
    \begin{itemize}
      \item \textbf{Armoiries officielles} du Royaume du Maroc intégrées en en-tête.
      \item Logo et coordonnées du cabinet d'avocats.
      \item Typographie \textbf{Noto Naskh Arabic} (RTL parfait).
      \item Statut mis à jour automatiquement : \texttt{finalized}.
    \end{itemize}
  \end{exampleblock}
\end{column}
\end{columns}
\end{frame}

  \begin{itemize}
    \item Entité centrale : \textbf{\texttt{Cabinet}} (Tenant).
    \item Clé étrangère \texttt{cabinet\_id} indexée et obligatoire sur toutes les tables (\texttt{cases}, \texttt{documents}, \texttt{drafts}).
    \item Cloisonnement systématique au niveau ORM : impossible d'accéder aux données d'un confrère.
  \end{itemize}

  \smallskip
  \textbf{Protection des identifiants :}
  \begin{itemize}
    \item Mots de passe hashés avec \textbf{bcrypt} + sel.
    \item Authentification par sessions JWT et cookies sécurisés HTTP-Only.
  \end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
  \begin{alertblock}{Audit BDD : Choix d'Adminer sécurisé}
    \begin{itemize}
      \item \textbf{Choix :} Conteneur \texttt{adminer:latest} sur le VPS (zéro installation locale, ultra-léger).
      \item \textbf{Secret professionnel (Art. 30 Loi 28-08) :} Port public 8080 bloqué, route Caddy neutralisée (404/308).
      \item \textbf{Accès chiffré exclusif :} Tunnel SSH via script \texttt{Ouvrir\_Adminer.bat} mappant \texttt{localhost:8088}.
    \end{itemize}
  \end{alertblock}
\end{column}
\end{columns}
\end{frame}
